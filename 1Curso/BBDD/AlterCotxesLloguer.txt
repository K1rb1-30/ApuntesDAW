DROP DATABASE IF EXISTS lloguer_cotxes;
CREATE DATABASE lloguer_cotxes CHARACTER SET utf8;
USE lloguer_cotxes;

CREATE TABLE cotxes (
    matricula VARCHAR(10) PRIMARY KEY,
    marca VARCHAR(50) NOT NULL,
    model VARCHAR(50) NOT NULL,
    color VARCHAR(30),
    classe ENUM('A', 'B', 'C') DEFAULT 'B',
    any_matriculacio  YEAR,
    tipus ENUM('furgoneta','turisme'),
    combustible ENUM('diesel','gasolina','electric'),
    quilometratge INT,
    preuDia INT NOT NULL CHECK (preuDia > 0),
    disponibilitat TINYINT(1)
);

CREATE TABLE clients (
    idClient INT AUTO_INCREMENT PRIMARY KEY,
    dni VARCHAR(9) NOT NULL UNIQUE,
    nom VARCHAR(50) NOT NULL,
    cognom1 VARCHAR(50) NOT NULL,
    cognom2 VARCHAR(50),
    telefon VARCHAR(15),
    email VARCHAR(100) ,
    adreca VARCHAR(150),
    data_naixement DATE,
    numero_carnet VARCHAR(20),
    data_caducitat DATE,
    CONSTRAINT ck_mail CHECK (email LIKE '%_@_%._%')
);

-- Taula resultat de relacio lloguer N-M 
CREATE TABLE lloguers (
    idLloguer INT PRIMARY KEY AUTO_INCREMENT,
    idClient INT,
    matricula VARCHAR(10),
    data_inici DATETIME,
    data_fi DATETIME,
    preu_total DECIMAL(6,2),
    CONSTRAINT ck_dates CHECK (data_inici < data_fi),
    CONSTRAINT uk_lloguers UNIQUE(matricula, data_inici), -- Es modificarà en activitat futura
    CONSTRAINT fk_lloguers_clients FOREIGN KEY (idClient) REFERENCES clients(idClient),
    CONSTRAINT fk_llogues_cotxes FOREIGN KEY (matricula) REFERENCES cotxes(matricula)
);
