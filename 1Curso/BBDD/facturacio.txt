DROP DATABASE IF EXISTS facturacio;

CREATE DATABASE facturacio CHARACTER SET utf8;

USE facturacio;

CREATE TABLE clients(
	NIF CHAR(9),
	Nom VARCHAR(35) NOT NULL,
	Cognom1 VARCHAR(35) NOT NULL,
	Cognom2 VARCHAR(35),
	Adreca VARCHAR(70) NOT NULL,
	DataAlta DATETIME DEFAULT CURRENT_TIMESTAMP(),
	CONSTRAINT pk_clients PRIMARY KEY (NIF)
);


CREATE TABLE productes(
	CodiProducte INT AUTO_INCREMENT,
	NomProducte VARCHAR(40) NOT NULL,
	Descripcio VARCHAR(20),
	PreuUnitat DECIMAL(6, 2) NOT NULL, -- Ponemos 6 para que haya 4 enteros y 2 decimales
	CONSTRAINT pk_productes PRIMARY KEY (CodiProducte)
);

CREATE TABLE factures(
	NumFactura INT(4),
	NIFClient CHAR(9),
	DataFactura DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP(),
	Total	INT	NOT NULL,
	CONSTRAINT pk_factures PRIMARY KEY (NumFactura),
	CONSTRAINT fk_client FOREIGN KEY (NIFClient) REFERENCES clients(NIF) ON UPDATE CASCADE ON DELETE RESTRICT
);

CREATE TABLE facturesProductes(
	NumFact INT(4),
	CodiProducte INT,
	Quantitat INT(5) NOT NULL,
	Total	INT	NOT NULL,
	DataFactura	DATETIME,
	CONSTRAINT pk_facturesProductes PRIMARY KEY (NumFact, CodiProducte),
	CONSTRAINT fk_factures FOREIGN KEY (NumFact) REFERENCES factures(NumFactura) ON UPDATE CASCADE ON DELETE CASCADE,
	CONSTRAINT fk_productes FOREIGN KEY (CodiProducte) REFERENCES productes(CodiProducte) ON UPDATE CASCADE ON DELETE RESTRICT,
	CONSTRAINT CHK_quantitat CHECK (Quantitat > 0)
);
